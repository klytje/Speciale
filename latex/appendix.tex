\section{Solving the scattering potential}
The starting point for this appendix is equation (\ref{theory: r-matrix: scattering matrix}), replicated here for convenience:
\begin{align*}
	U_l = \frac{I_l(\rho) - \rho R_l(E)I_l'(\rho)}{O_l(\rho) - \rho R_l(E)O_l'(\rho)} = e^{2i\phi_l}\frac{1 - L_l^* R_l(E)}{1 - L_l R_l(E)}
\end{align*}

We want to derive all the steps between the second and third term. \\
\begin{align*}
	I_l O_l' - I_l' O_l = 2i
\end{align*}

\begin{align*}
	\frac{I_l'}{I_l} = \frac{I_l O_l' - 2i}{O_l I_l} = \frac{O_l'}{O_l} - \frac{2i}{O_l I_l}
\end{align*}

\begin{align*}
		U_l &= \frac{I_l - \rho R_l I_l'}{O_l - \rho R_l O_l'} \\
		&= \frac{I_l(1 - I_l^2 \rho R_l \frac{I_l'}{I_l})}{O_l(1 - O_l^2 \rho R_l \frac{O_l'}{O_l})} \\
		&= \frac{I_l(1 - I_l^2 \rho R_l (\frac{O_l'}{O_l} - \frac{2i}{O_l I_l})}{O_l(1 - O_l^2 \rho R_l \frac{O_l'}{O_l})} \\
		&= \frac{I_l}{O_l}\frac{1 - I_l^2\rho R_l \frac{O_l'}{O_l} - 2i \rho R_l \frac{I_l}{O_l}}{1 - O_l^2 \rho R_l \frac{O_l'}{O_l}}
\end{align*}

\begin{align*}
	e^{2i\phi_l}\frac{1 - (\rho\frac{O_l'}{O_l})^* R_l(E)}{1 - \rho\frac{O_l'}{O_l} R_l(E)}
\end{align*}
